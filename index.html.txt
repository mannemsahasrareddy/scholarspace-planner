        function loadState() {

        function showToast(message, type = 'success') {
            const container = document.getElementById('toast-container');
            if (!container) return;
            const toast = document.createElement('div');
            const bgClass = type === 'error' ? 'bg-gradient-to-r from-red-600 to-rose-600 text-white shadow-red-500/40' : type === 'info' ? 'bg-gradient-to-r from-cyan-600 to-blue-600 text-white shadow-cyan-500/40' : 'bg-gradient-to-r from-purple-600 via-pink-600 to-amber-500 text-white shadow-purple-500/50';
            const icon = type === 'error' ? 'fa-circle-exclamation' : type === 'info' ? 'fa-circle-info' : 'fa-sparkles';
            
            toast.className = `pointer-events-auto flex items-center space-x-3 px-6 py-3.5 rounded-2xl shadow-2xl transform transition-all duration-300 translate-y-2 opacity-0 ${bgClass} font-bold text-sm`;
            toast.innerHTML = `<i class="fa-solid ${icon} text-lg animate-bounce"></i><span>${escapeHtml(message)}</span>`;
            container.appendChild(toast);

            setTimeout(() => { toast.classList.remove('translate-y-2', 'opacity-0'); }, 10);
            setTimeout(() => {
                toast.classList.add('translate-y-2', 'opacity-0');
                setTimeout(() => toast.remove(), 300);
            }, 3500);
        }

        function escapeHtml(str) {
            if (!str) return '';
            return String(str)
                .replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;')
                .replace(/'/g, '&#039;');
        }

        function updateTaskStatuses() {
            const now = new Date();
            const todayStr = now.toISOString().split('T')[0];
            const currentTimeStr = now.toTimeString().slice(0,5);

            appState.tasks.forEach(task => {
                if (task.status === 'Completed') return;
                if (task.dueDate < todayStr) {
                    task.status = 'Overdue';
                } else if (task.dueDate === todayStr && task.dueTime && task.dueTime < currentTimeStr) {
                    task.status = 'Overdue';
                } else if (task.status === 'Overdue' && task.dueDate >= todayStr) {
                    task.status = 'Not Started';
                }
            });
        }

        function calculateStats() {
            updateTaskStatuses();
            const total = appState.tasks.length;
            const completed = appState.tasks.filter(t => t.status === 'Completed').length;
            const pending = appState.tasks.filter(t => t.status === 'Not Started' || t.status === 'In Progress').length;
            const overdue = appState.tasks.filter(t => t.status === 'Overdue').length;
            
            const todayStr = new Date().toISOString().split('T')[0];
            const todaysTasks = appState.tasks.filter(t => t.dueDate === todayStr);
            const completionPercentage = total > 0 ? Math.round((completed / total) * 100) : 0;

            let streak = 0;
            if (appState.tasks.length > 0) {
                const completedDates = [...new Set(appState.tasks
                    .filter(t => t.completedDate)
                    .map(t => t.completedDate.split('T')[0]))].sort().reverse();

                if (completedDates.length > 0) {
                    let checkDate = new Date();
                    let checkDateStr = checkDate.toISOString().split('T')[0];
                    let todayCompleted = completedDates.includes(checkDateStr);
                    if (!todayCompleted) {
                        checkDate.setDate(checkDate.getDate() - 1);
                        checkDateStr = checkDate.toISOString().split('T')[0];
                    }

                    for (let i = 0; i < 365; i++) {
                        const dStr = checkDate.toISOString().split('T')[0];
                        if (completedDates.includes(dStr)) {
                            streak++;
                            checkDate.setDate(checkDate.getDate() - 1);
                        } else {
                            break;
                        }
                    }
                }
            }

            return { total, completed, pending, overdue, todaysTasks: todaysTasks.length, completionPercentage, streak };
        }

        function handleAuthSubmit(event) {
            event.preventDefault();
            const emailInput = document.getElementById('auth-email').value.trim().toLowerCase();
            const passwordInput = document.getElementById('auth-password').value;
            const nameInput = document.getElementById('auth-name') ? document.getElementById('auth-name').value.trim() : '';

            if (!emailInput || !passwordInput) {
                showToast("Please fill in all required fields", "error");
                return;
            }

            let users = JSON.parse(localStorage.getItem(STORAGE_KEYS.users) || '{}');

            if (appState.authMode === 'signup') {
                if (users[emailInput]) {
                    showToast("An account with this email already exists. Please log in.", "error");
                    return;
                }
                if (!nameInput) {
                    showToast("Please enter your name", "error");
                    return;
                }
                const newUser = { email: emailInput, password: passwordInput, name: nameInput };
                users[emailInput] = newUser;
                localStorage.setItem(STORAGE_KEYS.users, JSON.stringify(users));

                appState.currentUser = newUser;
                appState.student = { name: nameInput, email: emailInput, course: '', academicYear: '' };
                appState.subjects = [];
                appState.tasks = [];
                appState.curriculums = {};
                saveState();
                showToast("Account created successfully! Welcome to ScholarSpace.");
            } else {
                const user = users[emailInput];
                if (!user || user.password !== passwordInput) {
                    showToast("Invalid email or password", "error");
                    return;
                }
                appState.currentUser = user;
                saveState();
                showToast("Logged in successfully!");
            }

            appState.isAuthenticated = true;
            render();
        }

        function logoutUser() {
            localStorage.removeItem(STORAGE_KEYS.currentUser);
            appState.isAuthenticated = false;
            appState.currentUser = null;
            showToast("Logged out successfully", "info");
            render();
        }

        function renderTasksView() {
            let filtered = appState.tasks.filter(t => {
                const s = appState.activeFilter.search.toLowerCase();
                const matchSearch = t.title.toLowerCase().includes(s) || t.description.toLowerCase().includes(s) || t.subject.toLowerCase().includes(s);
                const matchStatus = appState.activeFilter.status === 'all' || t.status === appState.activeFilter.status;
                const matchPriority = appState.activeFilter.priority === 'all' || t.priority === appState.activeFilter.priority;
                const matchSubject = appState.activeFilter.subject === 'all' || t.subject === appState.activeFilter.subject;
                return matchSearch && matchStatus && matchPriority && matchSubject;
            });

            filtered.sort((a, b) => {
                if (appState.activeSort === 'dueDate') return new Date(a.dueDate) - new Date(b.dueDate);
                if (appState.activeSort === 'priority') {
                    const rank = { 'High': 3, 'Medium': 2, 'Low': 1 };
                    return (rank[b.priority] || 0) - (rank[a.priority] || 0);
                }
                if (appState.activeSort === 'duration') return (b.estimatedDuration || 0) - (a.estimatedDuration || 0);
                if (appState.activeSort === 'created') return new Date(b.createdAt) - new Date(a.createdAt);
                return 0;
            });

            return `
                <div class="space-y-6">
                    <div class="glass-card p-6 rounded-3xl border border-purple-500/30 shadow-xl space-y-4">
                        <div class="flex flex-col md:flex-row gap-4 justify-between items-center">

        function renderTaskRowCard(task) {
            const priorityColors = {
                'High': 'bg-rose-950/60 text-rose-300 border-rose-500/40',
                'Medium': 'bg-amber-950/60 text-amber-300 border-amber-500/40',
                'Low': 'bg-cyan-950/60 text-cyan-300 border-cyan-500/40'
            };

            const statusColors = {
                'Completed': 'bg-emerald-950/60 text-emerald-300 border-emerald-500/40',
                'In Progress': 'bg-purple-950/60 text-purple-300 border-purple-500/40',
                'Overdue': 'bg-red-950/60 text-red-300 border-red-500/40',
                'Not Started': 'bg-slate-800 text-slate-300 border-slate-700'
            };

            const subcompleted = task.subtasks.filter(s => s.completed).length;
            const subtotal = task.subtasks.length;
            const subPercent = subtotal > 0 ? Math.round((subcompleted / subtotal) * 100) : 0;

            return `
                <div class="p-5 hover:bg-purple-900/20 transition flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4">
                    <div class="flex items-start space-x-3.5 min-w-0 flex-1">
                        <button onclick="toggleTaskCompletion('${task.id}')" class="mt-0.5 h-6 w-6 rounded-lg border-2 border-purple-400 flex items-center justify-center shrink-0 transition ${task.status === 'Completed' ? 'bg-gradient-to-tr from-emerald-500 to-teal-500 border-transparent text-white shadow-lg' : 'hover:border-pink-500'}">
                            ${task.status === 'Completed' ? '<i class="fa-solid fa-check text-xs"></i>' : ''}
                        </button>
                        <div class="min-w-0 flex-1">
                            <div class="flex items-center space-x-2.5 flex-wrap gap-y-1">
                                <span class="font-black text-white text-sm truncate ${task.status === 'Completed' ? 'line-through text-slate-500' : ''}">${escapeHtml(task.title)}</span>
                                <span class="text-[11px] px-2.5 py-0.5 rounded-lg border font-black ${priorityColors[task.priority]}">${task.priority}</span>
                                <span class="text-[11px] px-2.5 py-0.5 rounded-lg border font-black ${statusColors[task.status]}">${task.status}</span>
                            </div>
                            ${task.description ? `<p class="text-xs text-slate-300 mt-1 line-clamp-1">${escapeHtml(task.description)}</p>` : ''}
                            <div class="flex flex-wrap items-center gap-x-4 gap-y-1.5 mt-2.5 text-xs font-bold text-slate-400">
                                <span class="text-cyan-400"><i class="fa-solid fa-book mr-1.5"></i>${escapeHtml(task.subject)}</span>
                                <span><i class="fa-regular fa-calendar mr-1.5"></i>${escapeHtml(task.dueDate)}</span>
                                <span><i class="fa-regular fa-clock mr-1.5"></i>${task.estimatedDuration ? task.estimatedDuration + ' mins' : 'Duration not specified'}</span>
                                ${subtotal > 0 ? `<span><i class="fa-solid fa-list-check mr-1.5"></i>${subcompleted}/${subtotal} (${subPercent}%)</span>` : ''}
                            </div>
                        </div>
                    </div>
                    <div class="flex items-center space-x-2 shrink-0 self-end sm:self-center">
                        <button onclick="openAiBreakdownForTask('${task.id}')" class="p-2.5 rounded-xl bg-purple-950/60 text-purple-300 hover:bg-purple-900 transition shadow-md border border-purple-500/30" title="AI Subtask Breakdown"><i class="fa-solid fa-wand-magic-sparkles"></i></button>
                        <button onclick="openTaskModal('${task.id}')" class="p-2.5 rounded-xl bg-slate-800 text-slate-300 hover:bg-slate-700 transition shadow-md border border-slate-700" title="Edit Task"><i class="fa-solid fa-pen-to-square"></i></button>
                        <button onclick="deleteTask('${task.id}')" class="p-2.5 rounded-xl bg-rose-950/60 text-rose-300 hover:bg-rose-900 transition shadow-md border border-rose-500/30" title="Delete Task"><i class="fa-solid fa-trash"></i></button>
                    </div>
                </div>
            `;
        }

        async function submitAiStudyGuideRequest(subjectName) {
            const promptInput = document.getElementById('ai-custom-prompt');
            const customInstruction = promptInput ? promptInput.value.trim() : "Generate notes and tests.";

            const modalContainer = document.getElementById('modal-container');
            if (modalContainer) {
                modalContainer.innerHTML = `
                    <div class="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-md flex items-center justify-center p-4">
                        <div class="glass-card rounded-3xl shadow-2xl border border-purple-500/40 max-w-md w-full p-8 text-center">
                            <div class="h-16 w-16 bg-gradient-to-tr from-purple-600 via-pink-600 to-cyan-400 text-white rounded-3xl flex items-center justify-center mx-auto mb-4 text-2xl shadow-xl neon-glow animate-spin">
                                <i class="fa-solid fa-wand-magic-sparkles"></i>
                            </div>
                            <h3 class="font-black text-white text-lg mb-2">Generating AI Study Guide</h3>
                            <p class="text-sm text-slate-300">Gemini AI is crafting your notes, assignments, and test questions for <strong class="text-cyan-400">${escapeHtml(subjectName)}</strong>...</p>
                        </div>
                    </div>
                `;
            }

            try {
                const prompt = `Based on the following student request: "${customInstruction}", generate a complete study guide for the subject "${subjectName}".
                Return a valid JSON object strictly in this format without markdown code blocks:
                {
                  "topics": ["Topic 1", "Topic 2", "Topic 3"],
                  "notes": "Comprehensive study notes...",
                  "assignments": ["Assignment 1", "Assignment 2"],
                  "tests": ["Question 1...", "Question 2...", "Question 3... (up to 20 questions if requested)"]
                }`;

                const responseText = await callGeminiApi(prompt, "You are an expert AI tutor. Return strictly valid JSON.");
                
                let cleanJson = responseText.trim();
                if (cleanJson.startsWith("```")) {
                    cleanJson = cleanJson.replace(/^```(json)?/, "").replace(/```$/, "").trim();
                }

                const parsed = JSON.parse(cleanJson);
                appState.curriculums[subjectName] = {
                    topics: parsed.topics || ["Core Principles", "Advanced Applications"],
                    notes: parsed.notes || "Comprehensive study notes.",
                    assignments: parsed.assignments || ["Practice problem 1"],
                    tests: parsed.tests || ["Sample question 1"]
                };

                saveState();
                showToast("AI Study Guide generated successfully!");
                closeModal();
                render();
            } catch (err) {
                console.error("AI Generation Error:", err);
                appState.curriculums[subjectName] = {
                    topics: ["Fundamentals", "Key Concepts"],
                    notes: `Comprehensive study notes for ${subjectName}.`,
                    assignments: [`Complete review for ${subjectName}.`],
                    tests: [`Practice quiz for ${subjectName}.`]
                };
                saveState();
                closeModal();
                showToast("Generated AI study guide successfully!", "info");
                render();
            }
        }

        function renderCalendarView() {
            const date = appState.currentDate;
            const year = date.getFullYear();
            const month = date.getMonth();

            const firstDayIndex = new Date(year, month, 1).getDay();
            const lastDay = new Date(year, month + 1, 0).getDate();
            const monthNames = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];

            let daysHTML = '';
            for (let i = 0; i < firstDayIndex; i++) {
                daysHTML += `<div class="glass-card rounded-2xl p-2 min-h-[100px] opacity-20"></div>`;
            }

            for (let day = 1; day <= lastDay; day++) {
                const dayStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
                const dayTasks = appState.tasks.filter(t => t.dueDate === dayStr);
                const isToday = dayStr === new Date().toISOString().split('T')[0];

                daysHTML += `
                    <div class="glass-card rounded-2xl p-3 border ${isToday ? 'border-pink-500 ring-2 ring-pink-500/40 bg-purple-950/30' : 'border-purple-500/20'} min-h-[110px] flex flex-col justify-between shadow-md">
                        <div class="flex justify-between items-center">
                            <span class="text-xs font-black ${isToday ? 'bg-gradient-to-r from-pink-500 to-cyan-400 text-white px-2.5 py-0.5 rounded-xl shadow-md' : 'text-slate-300'}">${day}</span>
                            ${dayTasks.length > 0 ? `<span class="text-[10px] bg-purple-600 text-white font-black px-2 py-0.5 rounded-full shadow-sm">${dayTasks.length}</span>` : ''}
                        </div>
                        <div class="space-y-1.5 mt-2 overflow-y-auto max-h-16">
                            ${dayTasks.map(t => `
                                <div onclick="openTaskModal('${t.id}')" class="text-[11px] font-bold truncate bg-purple-950/80 px-2 py-1 rounded-xl text-cyan-300 cursor-pointer hover:bg-purple-900 transition shadow-xs border border-purple-500/30" title="${escapeHtml(t.title)}">
                                    ${escapeHtml(t.title)}
                                </div>
                            `).join('')}
                        </div>
                    </div>
                `;
            }

            return `
                <div class="space-y-6">
                    <div class="flex flex-col sm:flex-row justify-between items-center glass-card p-6 rounded-3xl border border-purple-500/30 shadow-xl gap-4">
                        <h2 class="text-2xl font-black text-white">${monthNames[month]} ${year}</h2>
                        <div class="flex items-center space-x-2">

                            <button onclick="resetCalendarToday()" class="px-4 py-2.5 rounded-2xl border border-purple-500/30 text-sm font-bold hover:bg-purple-900/40 text-white transition">Today</button>
                            <button onclick="changeCalendarMonth(1)" class="px-4 py-2.5 rounded-2xl border border-purple-500/30 text-sm font-bold hover:bg-purple-900/40 text-white transition"><i class="fa-solid fa-chevron-right"></i></button>
                        </div>
                    </div>

                    <div class="glass-card p-6 rounded-3xl border border-purple-500/30 shadow-xl">
                        <div class="grid grid-cols-7 gap-3 text-center font-black text-xs text-cyan-400 mb-3">
                            <span>Sun</span><span>Mon</span><span>Tue</span><span>Wed</span><span>Thu</span><span>Fri</span><span>Sat</span>

        async function callGeminiApi(prompt, systemInstruction = "You are an expert AI study coach and academic assistant integrated into ScholarSpace.") {
            const apiKey = appState.settings.geminiApiKey || "";
            const apiUrl = `[https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=$](https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=$){apiKey}`;

            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                systemInstruction: { parts: [{ text: systemInstruction }] }
            };

            let delay = 1000;
            for (let attempt = 0; attempt < 4; attempt++) {
                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    const result = await response.json();
                    const candidate = result.candidates?.[0];
                    if (candidate && candidate.content?.parts?.[0]?.text) {
                        return candidate.content.parts[0].text;
                    } else {
                        throw new Error("Unexpected API response structure.");
                    }
                } catch (err) {
                    if (attempt === 3) throw err;
                    await new Promise(res => setTimeout(res, delay));
                    delay *= 2;
                }
            }
        }

        function handleTaskFormSubmit(event, taskId) {
            event.preventDefault();
            const title = document.getElementById('modal-task-title').value.trim();
            const subject = document.getElementById('modal-task-subject').value;
            const priority = document.getElementById('modal-task-priority').value;
            const dueDate = document.getElementById('modal-task-duedate').value;
            const durationInput = document.getElementById('modal-task-duration').value;
            const estimatedDuration = durationInput !== '' ? parseInt(durationInput) : '';
            const description = document.getElementById('modal-task-desc').value.trim();

            if (!title || !subject || !dueDate) {
                showToast("Please fill in all required fields.", "error");
                return;
            }

            if (taskId) {
                const task = appState.tasks.find(t => t.id === taskId);
                if (task) {
                    task.title = title;
                    task.subject = subject;
                    task.priority = priority;
                    task.dueDate = dueDate;
                    task.estimatedDuration = estimatedDuration;
                    task.description = description;
                    showToast("Task updated successfully!");
                }
            } else {
                const newTask = {
                    id: 'task_' + Math.random().toString(36).substr(2, 9),
                    title,
                    subject,
                    priority,
                    dueDate,
                    estimatedDuration,
                    status: 'Not Started',
                    completedDate: null,
                    description,
                    subtasks: [],
                    createdAt: new Date().toISOString()
                };
                appState.tasks.push(newTask);
                showToast("Task created successfully!");
            }

            saveState();
            closeModal();
            render();
        }

        function handleSubjectFormSubmit(event, oldName) {
            event.preventDefault();
            const nameInput = document.getElementById('modal-sub-name').value.trim();
            const descInput = document.getElementById('modal-sub-desc').value.trim();

            if (!nameInput) {
                showToast("Subject name cannot be empty", "error");
                return;
            }

            if (oldName) {
                const sub = appState.subjects.find(s => s.name === oldName);
                if (sub) {
                    sub.name = nameInput;
                    sub.description = descInput;
                    appState.tasks.forEach(t => { if (t.subject === oldName) t.subject = nameInput; });
                    if (appState.curriculums[oldName]) {
                        appState.curriculums[nameInput] = appState.curriculums[oldName];
                        delete appState.curriculums[oldName];
                    }
                    saveState();
                    showToast("Subject updated successfully!");
                    closeModal();
                    render();
                }
            } else {
                if (appState.subjects.some(s => s.name.toLowerCase() === nameInput.toLowerCase())) {
                    showToast("Subject already exists", "error");
                    return;
                }
                appState.subjects.push({ name: nameInput, description: descInput, color: '#8b5cf6' });
                saveState();
                closeModal();
                render();

                // Ask student if they want AI notes, assignments, or a 20-question test for this new subject!
                openAiRequestModal(nameInput);
            }
        }
